// Temporary stage harness (#1161). Appended to
// crates/secret-scan-core/src/detectors/connection_string.rs in a working copy,
// run with `cargo test --release -p redact-secret --lib bench_prose -- --ignored --nocapture`,
// then removed. Never committed under src/ (rust:check forbids std::time there).
#[cfg(test)]
mod zz_bench {
    use std::hint::black_box;
    use std::time::Instant;

    // The pre-#1161 collecting classifier (the shipped baseline).
    fn old_prose(value: &str) -> bool {
        const PREFIXES: [&str; 3] = ["insert", "replace", "todo"];
        let tokens: Vec<&str> = value
            .split(['_', '-'])
            .filter(|token| !token.is_empty())
            .collect();
        if tokens.len() < 2 {
            return false;
        }
        if tokens[tokens.len() - 1].eq_ignore_ascii_case("here") {
            return true;
        }
        PREFIXES
            .iter()
            .any(|prefix| tokens[0].eq_ignore_ascii_case(prefix))
            && tokens
                .iter()
                .any(|token| token.eq_ignore_ascii_case("password"))
    }

    fn timeit(f: fn(&str) -> bool, inputs: &[&str], reps: u32) -> f64 {
        let t = Instant::now();
        for _ in 0..reps {
            for s in inputs {
                black_box(f(black_box(s)));
            }
        }
        t.elapsed().as_nanos() as f64 / (f64::from(reps) * inputs.len() as f64)
    }
    fn med(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }
    fn compare(name: &str, inputs: &[&str]) {
        let fs: [fn(&str) -> bool; 3] = [old_prose, super::is_fill_in_password_prose, old_prose];
        let mut res: Vec<Vec<f64>> = vec![vec![]; 3];
        for round in 0..41 {
            for k in 0..3 {
                let idx = if round % 2 == 0 { k } else { 2 - k };
                res[idx].push(timeit(fs[idx], inputs, 20_000));
            }
        }
        let (b, n, a) = (med(res[0].clone()), med(res[1].clone()), med(res[2].clone()));
        eprintln!(
            "{name:24} old {b:8.2}ns | new {n:8.2} ({:+.1}%) | A/A {a:8.2} ({:+.1}%)",
            (n / b - 1.0) * 100.0,
            (a / b - 1.0) * 100.0
        );
    }

    #[test]
    #[ignore]
    fn bench_prose() {
        compare("secret-real", &["SYNTHETIC_REVOKED_DB_VALUE", "Zq9-Lm2_Xv7-Pk4"]);
        compare("placeholder-here", &["your_password_here", "root-password-HERE"]);
        compare("placeholder-prefix", &["REPLACE_ME_PASSWORD", "TODO_SET_PASSWORD"]);
        compare("single-token", &["SyntheticRevokedValue0123", "x9Kq2LmZ7vPd4"]);
        compare("separator-noise", &["__a--b__c--d__", "_-_-_-_-_-_-_-_-_"]);
        compare("long-30-tokens", &["a_b_c_d_e_f_g_h_i_j_k_l_m_n_o_p_q_r_s_t_u_v_w_x_y_z_aa_bb_cc_dd"]);
    }
}
