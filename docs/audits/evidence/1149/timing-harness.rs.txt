
#[cfg(test)]
mod zz_bench {
    use super::ValidationFailure as F;
    use std::hint::black_box;
    use std::time::Instant;
    type R = Result<(), F>;

    // The pre-#1149 validator (two modulo steps per letter).
    fn iban_old(c: &str) -> R {
        let bytes = c.as_bytes();
        if !(15..=34).contains(&bytes.len())
            || !bytes[0..2].iter().all(u8::is_ascii_uppercase)
            || !bytes[2..4].iter().all(u8::is_ascii_digit)
            || !bytes[4..]
                .iter()
                .all(|b| b.is_ascii_uppercase() || b.is_ascii_digit())
        {
            return Err(F::Malformed);
        }
        let mut r = 0u16;
        for b in bytes[4..].iter().chain(&bytes[..4]) {
            if b.is_ascii_digit() {
                r = super::push_decimal_digit(r, b - b'0');
            } else {
                let e = b - b'A' + 10;
                r = super::push_decimal_digit(r, e / 10);
                r = super::push_decimal_digit(r, e % 10);
            }
        }
        if r == 1 { Ok(()) } else { Err(F::ChecksumMismatch) }
    }
    fn iban_new(c: &str) -> R {
        super::validate_iban_mod97_v1(c)
    }

    fn timeit(f: fn(&str) -> R, inputs: &[&str], reps: u32) -> f64 {
        let t = Instant::now();
        for _ in 0..reps {
            for s in inputs {
                let _ = black_box(f(black_box(s)));
            }
        }
        t.elapsed().as_nanos() as f64 / (f64::from(reps) * inputs.len() as f64)
    }
    fn med(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }
    fn compare(name: &str, inputs: &[&str]) {
        let fs: [fn(&str) -> R; 3] = [iban_old, iban_new, iban_old];
        let mut res: Vec<Vec<f64>> = vec![vec![]; 3];
        for round in 0..41 {
            for k in 0..3 {
                let idx = if round % 2 == 0 { k } else { 2 - k };
                res[idx].push(timeit(fs[idx], inputs, 20_000));
            }
        }
        let base = med(res[0].clone());
        let new = med(res[1].clone());
        let aa = med(res[2].clone());
        eprintln!(
            "{name:22} old {base:7.2}ns | new {new:7.2} ({:+.1}%) | A/A {aa:7.2} ({:+.1}%)",
            (new / base - 1.0) * 100.0,
            (aa / base - 1.0) * 100.0
        );
    }

    #[test]
    #[ignore]
    fn bench_iban() {
        compare("iban22-mixed", &["GB82WEST12345698765432", "DE89370400440532013000"]);
        compare("iban31-alpha", &["MT84MALT011000012345MTLCAST001S"]);
        compare("iban34-alpha", &["ZZ25SYNTHETICABCDEFGHIJKLMNOPQRST", "ZZ25ABCDEFGHIJKLMNOPQRSTUVWXYZABCD"]);
        compare("iban34-numeric", &["ZZ25000000000000000000000000000000"]);
        compare("iban-checksum-bad", &["GB83WEST12345698765432"]);
        compare("iban-malformed-late", &["GB82WEST1234569876543x"]);
        compare("iban-malformed-early", &["gb82WEST12345698765432"]);
        compare("iban-short", &["GB82WEST"]);
    }
}
