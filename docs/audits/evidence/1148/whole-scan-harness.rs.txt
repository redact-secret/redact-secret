use redact_secret::{DefaultPolicy, DetectorRegistry, PiiSelection, scan};
use std::time::Instant;

fn text(lines: &[&str], n: usize, prefix: &str) -> String {
    let mut out = String::new();
    for i in 0..n {
        out.push_str(prefix);
        out.push_str(lines[i % lines.len()]);
        out.push_str(" and some ordinary prose follows here.\n");
    }
    out
}

#[test]
#[ignore]
fn whole_scan() {
    let selection = PiiSelection::parse(&["pii"]).unwrap();
    let registry = DetectorRegistry::with_built_in_and_pii(&selection).unwrap();
    let cases = [
        ("pan-valid", text(&["4242424242424242", "4111111111111111", "5555555555554444"], 2000, "card ")),
        ("pan-late-bad", text(&["4242424242424243", "4111111111111112"], 2000, "card ")),
        ("iban-mixed", text(&["GB82WEST12345698765432", "DE89370400440532013000", "FR1420041010050500013M02606"], 2000, "IBAN: ")),
        ("iban-alpha", text(&["MT84MALT011000012345MTLCAST001S", "BG80BNBG96611020345678"], 2000, "IBAN: ")),
        ("iban-bad-checksum", text(&["GB83WEST12345698765432", "DE90370400440532013000"], 2000, "IBAN: ")),
        ("prose", text(&["nothing here"], 2000, "")),
    ];
    for (name, input) in &cases {
        let mut v = Vec::new();
        let mut found = 0;
        for _ in 0..41 {
            let t = Instant::now();
            let f = scan(input, &registry, &DefaultPolicy).unwrap();
            found = f.len();
            v.push(t.elapsed().as_micros() as f64);
        }
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        eprintln!("WHOLE {name:18} bytes {:7} findings {found:5} median {:9.1}us min {:9.1}us", input.len(), v[20], v[0]);
    }
}
