//! Allocation and instruction probe for the incremental path (#1125).
//!
//! `inc_probe <workload> <chunk-bytes|whole> [passes]`: generates the named
//! `scan_cost` workload, builds the session outside the counted window, then
//! counts `GlobalAlloc` calls, requested bytes and peak live bytes of the
//! scan. Run it under `/usr/bin/time -l` for instructions retired; run it with
//! `none` as the workload to measure process start-up for subtraction.
#![allow(unsafe_code, clippy::all, clippy::pedantic, unused_imports, dead_code)]

use std::alloc::{GlobalAlloc, Layout, System};
use std::hint::black_box;
use std::sync::atomic::{AtomicUsize, Ordering::Relaxed};

use redact_secret::{
    DEFAULT_MAX_INPUT_BYTES, DefaultPolicy, DetectorRegistry, IncrementalLimits,
    IncrementalSanitizer, default_placeholder_formatter, scan_and_redact,
};

#[path = "../benches/support/workloads.rs"]
mod workloads;

struct Counting;
static CALLS: AtomicUsize = AtomicUsize::new(0);
static BYTES: AtomicUsize = AtomicUsize::new(0);
static LIVE: AtomicUsize = AtomicUsize::new(0);
static PEAK: AtomicUsize = AtomicUsize::new(0);

fn grow(n: usize) {
    let live = LIVE.fetch_add(n, Relaxed) + n;
    PEAK.fetch_max(live, Relaxed);
}
unsafe impl GlobalAlloc for Counting {
    unsafe fn alloc(&self, l: Layout) -> *mut u8 {
        CALLS.fetch_add(1, Relaxed);
        BYTES.fetch_add(l.size(), Relaxed);
        grow(l.size());
        unsafe { System.alloc(l) }
    }
    unsafe fn dealloc(&self, p: *mut u8, l: Layout) {
        LIVE.fetch_sub(l.size(), Relaxed);
        unsafe { System.dealloc(p, l) }
    }
    unsafe fn realloc(&self, p: *mut u8, l: Layout, new: usize) -> *mut u8 {
        CALLS.fetch_add(1, Relaxed);
        BYTES.fetch_add(new, Relaxed);
        LIVE.fetch_sub(l.size(), Relaxed);
        grow(new);
        unsafe { System.realloc(p, l, new) }
    }
}
#[global_allocator]
static A: Counting = Counting;

fn main() {
    let mut args = std::env::args().skip(1);
    let name = args.next().unwrap_or_else(|| "none".to_owned());
    if name == "none" {
        return;
    }
    let mode = args.next().unwrap_or_else(|| "whole".to_owned());
    let passes: usize = args.next().map_or(1, |p| p.parse().unwrap());
    let workload = workloads::WORKLOADS
        .iter()
        .find(|w| w.name == name)
        .expect("workload");
    let input = (workload.generate)();
    let registry = DetectorRegistry::with_built_in([]).unwrap();
    for _ in 0..passes {
        let sink;
        let before = (LIVE.load(Relaxed), 0);
        if mode == "whole" {
            CALLS.store(0, Relaxed);
            BYTES.store(0, Relaxed);
            PEAK.store(before.0, Relaxed);
            let result = scan_and_redact(
                &input,
                &registry,
                &DefaultPolicy,
                &default_placeholder_formatter,
            )
            .unwrap();
            sink = result.text().len() + result.findings().len();
        } else {
            let chunk: usize = mode.parse().unwrap();
            let chunks = workloads::partition(&input, chunk);
            let limits = IncrementalLimits::new(
                DEFAULT_MAX_INPUT_BYTES.max(input.len()),
                IncrementalLimits::minimum_buffered_bytes(
                    workloads::CLI_MAX_TOKEN_BYTES,
                    workloads::CLI_MAX_MULTILINE_BYTES,
                ),
                workloads::CLI_MAX_TOKEN_BYTES,
                workloads::CLI_MAX_MULTILINE_BYTES,
            )
            .unwrap();
            let mut session = IncrementalSanitizer::new(limits).unwrap();
            let live = LIVE.load(Relaxed);
            CALLS.store(0, Relaxed);
            BYTES.store(0, Relaxed);
            PEAK.store(live, Relaxed);
            let mut total = 0;
            for chunk in &chunks {
                let result = black_box(session.append(chunk).unwrap());
                total += result.text().len() + result.findings().len();
            }
            let result = session.finalize().unwrap();
            sink = total + result.text().len() + result.findings().len();
        }
        println!(
            "{name} {mode}: alloc_calls {} requested_bytes {} peak_live_above_start {} sink {sink}",
            CALLS.load(Relaxed),
            BYTES.load(Relaxed),
            PEAK.load(Relaxed).saturating_sub(before.0),
        );
    }
}
