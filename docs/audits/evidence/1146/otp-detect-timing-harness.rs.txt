
#[cfg(test)]
mod measure_otp {
    //! Timing harness (maintainer-only, not committed).
    use super::*;
    use std::hint::black_box;
    use std::time::Instant;

    const URI: &str = "otpauth://totp/Example:alice@example.com?secret=JBSWY3DPEHPK3PXP&issuer=Example";

    fn filler(n: usize) -> String {
        let mut s = String::new();
        while s.len() < n {
            s.push_str("The quick brown fox, request id 12345, status ok; path /var/log/app.log\n");
        }
        s
    }
    fn workloads() -> Vec<(&'static str, String)> {
        let mut sparse = filler(74_000);
        sparse.push_str(URI);
        let mut uni = filler(74_000);
        uni.push_str(" \u{d55c}\u{ae00}\u{1f600} ");
        uni.push_str(URI);
        let dense = format!("{URI}\n").repeat(1_000);
        let noise = "otpauth://push/x otpauth:/ otpauth://tot ".repeat(1_800);
        let denseuni = format!("\u{d55c}{URI}\n").repeat(1_000);
        let lead = "ooo otpaut otpauth: otpauth:/ oauth ".repeat(2_200);
        vec![
            ("otp-lead-noise", lead),
            ("otp-sparse-ascii", sparse),
            ("otp-sparse-unicode", uni),
            ("otp-dense", dense),
            ("otp-dense-unicode", denseuni),
            ("otp-prefix-noise", noise),
        ]
    }
    fn time(input: &str) -> f64 {
        let ctx = DetectorContext::new(input.len());
        let mut reps = 1usize;
        loop {
            let t = Instant::now();
            for _ in 0..reps {
                black_box(OtpauthDetector.detect(black_box(input), &ctx).unwrap());
            }
            if t.elapsed().as_millis() >= 10 || reps >= 100_000 {
                break;
            }
            reps *= 2;
        }
        let mut samples = Vec::new();
        for _ in 0..21 {
            let t = Instant::now();
            for _ in 0..reps {
                black_box(OtpauthDetector.detect(black_box(input), &ctx).unwrap());
            }
            samples.push(t.elapsed().as_nanos() as f64 / reps as f64 / 1000.0);
        }
        samples.sort_by(|a, b| a.partial_cmp(b).unwrap());
        samples[samples.len() / 2]
    }
    #[test]
    #[ignore = "timing harness"]
    fn measure_otp_detect() {
        for (name, input) in workloads() {
            eprintln!("RESULT {name} {:.3}", time(&input));
        }
    }
}
