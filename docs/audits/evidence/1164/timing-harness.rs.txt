#[cfg(test)]
mod zz_bench_1164 {
    use super::*;
    use crate::detectors::pattern::find_literal;
    use std::hint::black_box;
    use std::time::Instant;

    fn new(input: &str) -> usize {
        JwtDetector
            .detect(input, &DetectorContext::new(input.len()))
            .unwrap()
            .len()
    }

    // The pre-#1164 per-byte scan (same as the test oracle).
    fn old(input: &str) -> usize {
        let bytes = input.as_bytes();
        let mut count = 0usize;
        let mut cursor = 0usize;
        while cursor + HEADER_LEAD.len() <= bytes.len() {
            if &bytes[cursor..cursor + HEADER_LEAD.len()] != HEADER_LEAD.as_bytes() {
                cursor += 1;
                continue;
            }
            let Some(JwtMatch { end, payload }) = match_jwt_at(bytes, cursor) else {
                cursor += 1;
                continue;
            };
            let before_is_token = cursor > 0 && is_token_byte(bytes[cursor - 1]);
            let after_is_token = end < bytes.len() && is_token_byte(bytes[end]);
            if !before_is_token
                && !after_is_token
                && !is_supabase_legacy_anon_claim(&bytes[payload])
                && ByteRange::new(cursor, end).is_some()
            {
                count += 1;
            }
            cursor = end;
        }
        count
    }

    // Variant A: `find_literal` per search (lead table rebuilt each call).
    fn variant_a(input: &str) -> usize {
        let bytes = input.as_bytes();
        let mut count = 0usize;
        let mut cursor = 0usize;
        while let Some(at) = find_literal(bytes, HEADER_LEAD.as_bytes(), cursor) {
            cursor = at;
            let Some(JwtMatch { end, payload }) = match_jwt_at(bytes, cursor) else {
                cursor += 1;
                continue;
            };
            let before_is_token = cursor > 0 && is_token_byte(bytes[cursor - 1]);
            let after_is_token = end < bytes.len() && is_token_byte(bytes[end]);
            if !before_is_token
                && !after_is_token
                && !is_supabase_legacy_anon_claim(&bytes[payload])
                && ByteRange::new(cursor, end).is_some()
            {
                count += 1;
            }
            cursor = end;
        }
        count
    }

    fn timeit(f: fn(&str) -> usize, input: &str, reps: u32) -> f64 {
        let t = Instant::now();
        for _ in 0..reps {
            black_box(f(black_box(input)));
        }
        t.elapsed().as_nanos() as f64 / f64::from(reps)
    }
    fn med(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }
    fn compare(name: &str, input: &str, reps: u32) {
        assert_eq!(old(input), new(input));
        assert_eq!(old(input), variant_a(input));
        let fs: [fn(&str) -> usize; 4] = [old, variant_a, new, old];
        let mut res: Vec<Vec<f64>> = vec![vec![]; 4];
        for round in 0..31 {
            for k in 0..4 {
                let idx = if round % 2 == 0 { k } else { 3 - k };
                res[idx].push(timeit(fs[idx], input, reps));
            }
        }
        let base = med(res[0].clone());
        let a = med(res[1].clone());
        let newm = med(res[2].clone());
        let aa = med(res[3].clone());
        eprintln!(
            "{name:20} bytes {:7} old {base:10.1}ns | A {a:10.1} ({:+.1}%) | B(prod) {newm:10.1} ({:+.1}%) | A/A {aa:10.1} ({:+.1}%)",
            input.len(),
            (a / base - 1.0) * 100.0,
            (newm / base - 1.0) * 100.0,
            (aa / base - 1.0) * 100.0
        );
    }

    #[test]
    #[ignore]
    fn bench_jwt() {
        let jwt = "eyJTYNTH.eyJSYNTHETIC.SYNTHETIC_REVOKED_SIGNATURE";
        let filler = "The quick brown fox, request id 12345, status ok; path /var/log/app.log\n";
        let uni = "\u{d55c}\u{ae00} \u{1f600} synthetic text\n";
        compare(
            "sparse-ascii",
            &format!("{}{jwt}\n{}", filler.repeat(500), filler.repeat(500)),
            500,
        );
        compare(
            "sparse-unicode",
            &format!("{}{jwt}\n{}", uni.repeat(750), uni.repeat(750)),
            500,
        );
        compare("dense", &format!("{jwt}\n").repeat(1500), 300);
        compare("dense-short-gap", &format!("{jwt} ").repeat(1500), 300);
        compare("prefix-noise", &"eyJ eyJ. eyJabc.eyJ ".repeat(3000), 300);
        compare("eyJ-header-only", &"eyJTYNTH.".repeat(8000), 300);
        compare("no-match", &filler.repeat(1000), 500);
        compare("one-long-run", &format!("eyJ{}", "A".repeat(100_000)), 300);
    }
}
