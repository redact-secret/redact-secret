#![allow(clippy::unwrap_used, clippy::expect_used, clippy::panic)]
use redact_secret::{DefaultPolicy, DetectorRegistry, scan};
use std::time::Instant;

fn lines(values: &[&str], n: usize, name: &str, prose_every: usize) -> String {
    let mut out = String::new();
    for i in 0..n {
        if prose_every == 0 || i % prose_every == 0 {
            out.push_str(name);
            out.push('=');
            out.push_str(values[i % values.len()]);
            out.push('\n');
        } else {
            out.push_str("an ordinary line of prose that mentions nothing of note here.\n");
        }
    }
    out
}

#[test]
#[ignore]
fn whole_scan() {
    let registry = DetectorRegistry::with_built_in([]).unwrap();
    let placeholders = [
        "ntn_yourinternalintegrationtokenhere",
        "whsec_YOURSIGNINGSECRET",
        "sk_yourapikeyplaceholder",
    ];
    let alphabetic = [
        "ntn_generalpurposeapplicationname",
        "pk_quickbrownfoxjumpsoverlazydog",
        "sk_thequickbrownfoxjumpedoverit",
    ];
    let ambiguous = [
        "ntn_yourtokentokentokenx",
        "pk_yourkeykeykeykeykeykeyz",
        "sk_yourapikeyapikeyapikeyq",
    ];
    let long = [
        "ntn_yourtokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokentokenx",
    ];
    let jwt = "eyJTYNTH.eyJSYNTHETIC.SYNTHETIC_REVOKED_SIGNATURE";
    let filler = "The quick brown fox, request id 12345, status ok; path /var/log/app.log\n";
    let sparse_ascii = format!("{}{jwt}\n{}", filler.repeat(1000), filler.repeat(1000));
    let sparse_unicode = format!(
        "{}{jwt}\n{}",
        "\u{d55c}\u{ae00} \u{1f600} synthetic text\n".repeat(1500),
        "\u{d55c}\u{ae00} \u{1f600} synthetic text\n".repeat(1500)
    );
    let dense_jwt = format!("{jwt}\n").repeat(1500);
    let prefix_noise = "eyJ eyJ. eyJabc.eyJ ".repeat(3000);
    let cases = [
        ("jwt-sparse-ascii", sparse_ascii),
        ("jwt-sparse-unicode", sparse_unicode),
        ("jwt-dense", dense_jwt),
        ("jwt-prefix-noise", prefix_noise),
        (
            "sparse-prose",
            lines(&placeholders, 2000, "NOTION_TOKEN", 50),
        ),
        (
            "dense-placeholder",
            lines(&placeholders, 2000, "NOTION_TOKEN", 0),
        ),
        (
            "dense-alphabetic",
            lines(&alphabetic, 2000, "NOTION_TOKEN", 0),
        ),
        (
            "dense-ambiguous",
            lines(&ambiguous, 2000, "NOTION_TOKEN", 0),
        ),
        ("dense-long", lines(&long, 500, "NOTION_TOKEN", 0)),
        (
            "prose-only",
            lines(&placeholders, 2000, "NOTION_TOKEN", 1_000_000),
        ),
    ];
    for (name, input) in &cases {
        let mut v = Vec::new();
        let mut found = 0;
        for _ in 0..61 {
            let t = Instant::now();
            let f = scan(input, &registry, &DefaultPolicy).unwrap();
            found = f.len();
            v.push(t.elapsed().as_micros() as f64);
        }
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        eprintln!(
            "WHOLE {name:18} bytes {:7} findings {found:5} median {:9.1}us min {:9.1}us",
            input.len(),
            v[30],
            v[0]
        );
    }
}
