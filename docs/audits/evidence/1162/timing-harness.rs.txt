#[cfg(test)]
mod zz_bench_1162 {
    use super::{is_glued_instructional_placeholder, is_glued_instructional_placeholder_heap};
    use std::hint::black_box;
    use std::time::Instant;

    fn timeit(f: fn(&str) -> bool, input: &str, reps: u32) -> f64 {
        let t = Instant::now();
        for _ in 0..reps {
            black_box(f(black_box(input)));
        }
        t.elapsed().as_nanos() as f64 / f64::from(reps)
    }
    fn med(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }
    fn compare(name: &str, input: &str, reps: u32) {
        // old, new, old (A/A)
        let fs: [fn(&str) -> bool; 3] = [
            is_glued_instructional_placeholder_heap,
            is_glued_instructional_placeholder,
            is_glued_instructional_placeholder_heap,
        ];
        let mut res: Vec<Vec<f64>> = vec![vec![]; 3];
        for round in 0..41 {
            for k in 0..3 {
                let idx = if round % 2 == 0 { k } else { 2 - k };
                res[idx].push(timeit(fs[idx], input, reps));
            }
        }
        let base = med(res[0].clone());
        let new = med(res[1].clone());
        let aa = med(res[2].clone());
        eprintln!(
            "{name:24} len {:4} old {base:9.2}ns | new {new:9.2} ({:+.1}%) | A/A {aa:9.2} ({:+.1}%)",
            input.len(),
            (new / base - 1.0) * 100.0,
            (aa / base - 1.0) * 100.0
        );
    }

    #[test]
    #[ignore]
    fn bench_glued() {
        compare(
            "real-alphabetic",
            "yourgeneralpurposeapplicationname",
            200_000,
        );
        compare("placeholder", "yourapikeyplaceholder", 100_000);
        compare("placeholder-accept", "YourSecretTokenValue", 100_000);
        compare(
            "ambiguous",
            &format!("your{}x", "tokentoken".repeat(2)),
            50_000,
        );
        compare(
            "accept-medium",
            &format!("your{}", "apikey".repeat(10)),
            20_000,
        );
        compare(
            "accept-at-bound",
            &format!("your{}", "key".repeat(84)),
            5_000,
        );
        compare(
            "long-fallback",
            &format!("your{}", "key".repeat(100)),
            5_000,
        );
        compare("not-alpha", "your_api_key", 1_000_000);
        compare("no-lead", "generalapplicationname", 1_000_000);
    }
}
