+    fn time(f: &dyn Fn() -> usize) -> f64 {
+        let mut reps = 1usize;
+        loop {
+            let t = Instant::now();
+            let mut sink = 0;
+            for _ in 0..reps {
+                sink += std::hint::black_box(f());
+            }
+            let e = t.elapsed();
+            std::hint::black_box(sink);
+            if e.as_millis() >= 20 || reps >= 1_000_000 {
+                return e.as_secs_f64() * 1e6 / reps as f64;
+            }
+            reps *= 2;
+        }
+    }
+    fn median(mut v: Vec<f64>) -> f64 {
+        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
+        v[v.len() / 2]
