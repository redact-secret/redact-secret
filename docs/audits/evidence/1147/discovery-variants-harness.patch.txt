+fn detect_match_indices(input: &str, is_nfc: impl Fn(&str) -> bool) -> Vec<Alternative> {
+    let mut output = Vec::new();
+    for (at, _) in input.match_indices('@') {
+        let Some(start) = scan_local_start(input, at) else {
+            continue;
+        };
+        let Some(end) = scan_domain_end(input, at + 1) else {
+            continue;
+        };
+        if start == at || end == at + 1 {
+            continue;
+        }
+        let candidate = &input[start..end];
+        if candidate.len() > MAX_CANDIDATE_BYTES
+            || !valid_boundary(input, start, end)
+            || unsupported_enclosing_syntax(input, start, at, end)
+            || is_uri_userinfo(input, start, at)
+        {
+            continue;
+        }
+        let local = &input[start..at];
+        let domain = &input[at + 1..end];
+        if !valid_local(local) || !valid_domain(domain) || !is_nfc(candidate) {
+            continue;
+        }
+        let start = match email_label_key_length(local) {
+            Some(label) if valid_local(&local[label..]) => start + label,
+            Some(_) => continue,
+            None => start,
+        };
+        let sensitivity = if reserved_documentation_domain(domain) {
+            SensitivityState::NonSensitive
+        } else {
+            SensitivityState::NotEstablished
+        };
+        let Some(range) = ByteRange::new(start, end) else {
+            continue;
+        };
+        output.push(Alternative {
+            family_id: FAMILY_ID,
+            domain: IdentityDomain::Email,
+            range,
+            identity: IdentityState::Established,
+            identity_confidence: Confidence::High,
+            identity_specificity: Specificity::Structural,
+            sensitivity,
+            sensitivity_confidence: Confidence::High,
+            sensitivity_specificity: Specificity::Contextual,
+            obfuscation: Obfuscation::None,
+            reject_invisible_normalization: true,
+        });
+    }
+    output

+    /// Scalar probe of a short window, then a memchr-backed jump when the
+    /// window had no `@`; the probe resumes after every hit.
+    struct AtPositions<'a> {
+        input: &'a str,
+        pos: usize,
+    }
+    impl<'a> AtPositions<'a> {
+        fn new(input: &'a str) -> Self {
+            Self { input, pos: 0 }
+        }
+    }
+    impl Iterator for AtPositions<'_> {
+        type Item = usize;
+        fn next(&mut self) -> Option<usize> {
+            const PROBE: usize = 64;
+            let bytes = self.input.as_bytes();
+            while self.pos < bytes.len() {
+                let probe_end = (self.pos + PROBE).min(bytes.len());
+                if let Some(offset) = bytes[self.pos..probe_end].iter().position(|b| *b == b'@') {
+                    let at = self.pos + offset;
+                    self.pos = at + 1;
+                    return Some(at);
+                }
+                let mut jump = probe_end;
+                while !self.input.is_char_boundary(jump) {
+                    jump += 1;
+                }
+                match self.input[jump..].find('@') {
+                    Some(offset) => {
+                        let at = jump + offset;
+                        self.pos = at + 1;
+                        return Some(at);
+                    }
+                    None => {
+                        self.pos = bytes.len();
+                    }
+                }
+            }
+            None
+        }
+    }

+    fn time(f: &dyn Fn() -> usize) -> f64 {
+        let mut reps = 1usize;
+        loop {
+            let t = Instant::now();
+            let mut sink = 0;
+            for _ in 0..reps {
+                sink += std::hint::black_box(f());
+            }
+            let e = t.elapsed();
+            std::hint::black_box(sink);
+            if e.as_millis() >= 20 || reps >= 1_000_000 {
+                return e.as_secs_f64() * 1e6 / reps as f64;
+            }
+            reps *= 2;
+        }
+    }
