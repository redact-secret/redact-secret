// Temporary measurement harness for #1169 (NOT compiled, never under src in a
// commit). Append to crates/secret-scan-core/src/pii.rs (it needs the private
// `pii_us_ssn` / `pii_payment_card` modules), run
//   cargo test -p redact-secret --release --lib measure_1169 -- --ignored --nocapture
// then remove it again. It uses clock names, which scripts/check-rust-workspace.py
// forbids under src.
//
// Also make `detect_us_ssns`, `detect_us_ssns_with`, `detect_payment_cards`
// and `detect_payment_cards_with` temporarily `pub(super)` so this module can
// call them; revert that with the harness.
//
// Baseline A is the detector run with the one-byte-at-a-time walk
// (`scalar`), which is the pre-change loop head moved behind the discovery
// parameter.

#[cfg(test)]
mod timing_1169 {
    use super::*;
    use std::time::Instant;

    fn scalar(bytes: &[u8], from: usize) -> usize {
        let mut at = from.min(bytes.len());
        while at < bytes.len() && !bytes[at].is_ascii_digit() {
            at += 1;
        }
        at
    }

    /// The issue's first naive prototype: a nearby digit among the first four
    /// bytes, else skip the whole 16-byte block when the other twelve are
    /// digit-free.
    fn naive(bytes: &[u8], from: usize) -> usize {
        let mut at = from.min(bytes.len());
        while at < bytes.len() {
            if bytes[at].is_ascii_digit() {
                return at;
            }
            if let Some(block) = bytes.get(at..at + 16)
                && !block[..4].iter().any(u8::is_ascii_digit)
                && !block[4..].iter().any(u8::is_ascii_digit)
            {
                at += 16;
                continue;
            }
            at += 1;
        }
        at
    }

    fn adaptive<const PROBE: usize, const BLOCK: usize>(bytes: &[u8], from: usize) -> usize {
        let len = bytes.len();
        let mut at = from.min(len);
        let probe_end = at.saturating_add(PROBE).min(len);
        while at < probe_end {
            if bytes[at].is_ascii_digit() {
                return at;
            }
            at += 1;
        }
        while let Some(block) = bytes.get(at..at + BLOCK) {
            if block.iter().fold(false, |found, byte| found | byte.is_ascii_digit()) {
                break;
            }
            at += BLOCK;
        }
        while at < len && !bytes[at].is_ascii_digit() {
            at += 1;
        }
        at
    }

    // Generic over the discovery, so each variant is a distinct monomorphic
    // (inlinable) instantiation, as the shipped `next_ascii_digit` is.
    fn ssn<F: Fn(&[u8], usize) -> usize>(input: &str, f: F) -> usize {
        pii_us_ssn::detect_us_ssns_with(input, f).len()
    }
    fn pan<F: Fn(&[u8], usize) -> usize>(input: &str, f: F) -> usize {
        pii_payment_card::detect_payment_cards_with(input, f).len()
    }

    struct Lcg(u64);
    impl Lcg {
        fn next(&mut self, bound: u64) -> u64 {
            self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
            (self.0 >> 33) % bound
        }
    }

    fn workloads(kind: &str) -> Vec<(String, String)> {
        let filler = "The quick brown fox jumps over the lazy dog while the build runs. ";
        let ufiller = "고객 안내문입니다 \u{1f600} 계속 읽어 주세요. ";
        let target = if kind == "ssn" { "ssn: 078-05-1120 " } else { "card: 4000 0087 7000 0003 " };
        let sparse = |f: &str, n: usize| {
            let mut s = f.repeat(n);
            let mid = s.len() / 2;
            let mid = (mid..).find(|i| s.is_char_boundary(*i)).unwrap();
            s.insert_str(mid, target);
            s
        };
        let mut rng = Lcg(0x9E3779B97F4A7C15);
        let mut noise = String::new();
        while noise.len() < 48_000 {
            for _ in 0..(1 + rng.next(11)) {
                noise.push(char::from(b'0' + rng.next(10) as u8));
            }
            noise.push(if rng.next(3) == 0 { '-' } else { ' ' });
        }
        // Isolated digits every `gap` bytes: each defeats a block skip.
        let isolated = |gap: usize| {
            let mut s = String::new();
            while s.len() < 60_000 {
                s.push_str(&"x".repeat(gap - 1));
                s.push('7');
            }
            s
        };
        let dense = if kind == "ssn" {
            "123-45-6789 ".repeat(1000)
        } else {
            "4000 0087 7000 0003 ".repeat(850)
        };
        vec![
            ("sparse ascii 74KB".into(), sparse(filler, 1100)),
            ("sparse unicode 78KB".into(), sparse(ufiller, 1500)),
            ("dense".into(), dense),
            ("digit noise 48KB".into(), noise),
            ("isolated digit /17".into(), isolated(17)),
            ("isolated digit /24".into(), isolated(24)),
            ("isolated digit /40".into(), isolated(40)),
            ("long digit run 60KB".into(), "9".repeat(60_000)),
            ("all non-digit 60KB".into(), "ab-cd ".repeat(10_000)),
        ]
    }

    fn time(f: &dyn Fn() -> usize) -> f64 {
        let mut reps = 1u32;
        loop {
            let t = Instant::now();
            let mut sink = 0;
            for _ in 0..reps {
                sink += std::hint::black_box(f());
            }
            let e = t.elapsed();
            std::hint::black_box(sink);
            if e.as_millis() >= 20 || reps >= 1_000_000 {
                return e.as_secs_f64() * 1e6 / f64::from(reps);
            }
            reps *= 2;
        }
    }
    fn median(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }

    #[test]
    #[ignore]
    fn measure_1169() {
        for kind in ["ssn", "pan"] {
            for (name, input) in workloads(kind) {
                let is_ssn = kind == "ssn";
                fn run<F: Fn(&[u8], usize) -> usize>(is_ssn: bool, input: &str, f: F) -> usize {
                    if is_ssn { ssn(input, f) } else { pan(input, f) }
                }
                let expect = run(is_ssn, &input, scalar);
                assert_eq!(expect, run(is_ssn, &input, naive));
                assert_eq!(expect, run(is_ssn, &input, adaptive::<16, 16>));
                assert_eq!(expect, run(is_ssn, &input, adaptive::<8, 16>));
                assert_eq!(expect, run(is_ssn, &input, adaptive::<32, 32>));
                let i = input.as_str();
                let production = move || {
                    if is_ssn {
                        pii_us_ssn::detect_us_ssns(i).len()
                    } else {
                        pii_payment_card::detect_payment_cards(i).len()
                    }
                };
                let variants: Vec<(&str, Box<dyn Fn() -> usize + '_>)> = vec![
                    ("A  scalar (pre-change)", Box::new(move || run(is_ssn, i, scalar))),
                    ("A2 scalar (A/A control)", Box::new(move || run(is_ssn, i, scalar))),
                    ("A3 scalar (second control)", Box::new(move || run(is_ssn, i, scalar))),
                    ("N  naive block skip", Box::new(move || run(is_ssn, i, naive))),
                    ("P8B16 probe8+block16", Box::new(move || run(is_ssn, i, adaptive::<8, 16>))),
                    ("P16B16 probe16+block16", Box::new(move || run(is_ssn, i, adaptive::<16, 16>))),
                    ("P32B32 probe32+block32", Box::new(move || run(is_ssn, i, adaptive::<32, 32>))),
                    ("P  production", Box::new(production)),
                ];
                let mut samples: Vec<Vec<f64>> = vec![Vec::new(); variants.len()];
                for round in 0..61 {
                    let order: Vec<usize> = if round % 2 == 0 {
                        (0..variants.len()).collect()
                    } else {
                        (0..variants.len()).rev().collect()
                    };
                    for k in order {
                        samples[k].push(time(&variants[k].1));
                    }
                }
                let base = median(samples[0].clone());
                let base_min = samples[0].iter().copied().fold(f64::INFINITY, f64::min);
                for (k, (label, _)) in variants.iter().enumerate() {
                    let m = median(samples[k].clone());
                    let lo = samples[k].iter().copied().fold(f64::INFINITY, f64::min);
                    println!(
                        "{kind} {name:22} {label:26} median {m:9.2} us {:+7.1}%   min {lo:9.2} us {:+7.1}%",
                        (m / base - 1.0) * 100.0,
                        (lo / base_min - 1.0) * 100.0
                    );
                }
            }
        }
    }
}
