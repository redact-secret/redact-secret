// Temporary production-only harness for #1169 (NOT compiled). It times only
// `detect_us_ssns` / `detect_payment_cards`, so the same text can be appended
// to the pre-change and the post-change `pii.rs` (with those two functions
// made `pub(super)`) and the two release test binaries compared process
// against process (alternating runs).

#[cfg(test)]
mod timing_1169_production {
    use super::*;
    use std::time::Instant;

    struct Lcg(u64);
    impl Lcg {
        fn next(&mut self, bound: u64) -> u64 {
            self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
            (self.0 >> 33) % bound
        }
    }

    fn workloads(kind: &str) -> Vec<(String, String)> {
        let filler = "The quick brown fox jumps over the lazy dog while the build runs. ";
        let ufiller = "고객 안내문입니다 \u{1f600} 계속 읽어 주세요. ";
        let target = if kind == "ssn" { "ssn: 078-05-1120 " } else { "card: 4000 0087 7000 0003 " };
        let sparse = |f: &str, n: usize| {
            let mut s = f.repeat(n);
            let mid = s.len() / 2;
            let mid = (mid..).find(|i| s.is_char_boundary(*i)).unwrap();
            s.insert_str(mid, target);
            s
        };
        let mut rng = Lcg(0x9E3779B97F4A7C15);
        let mut noise = String::new();
        while noise.len() < 48_000 {
            for _ in 0..(1 + rng.next(11)) {
                noise.push(char::from(b'0' + rng.next(10) as u8));
            }
            noise.push(if rng.next(3) == 0 { '-' } else { ' ' });
        }
        // Isolated digits every `gap` bytes: each defeats a block skip.
        let isolated = |gap: usize| {
            let mut s = String::new();
            while s.len() < 60_000 {
                s.push_str(&"x".repeat(gap - 1));
                s.push('7');
            }
            s
        };
        let dense_unit = if kind == "ssn" { "123-45-6789 " } else { "4000 0087 7000 0003 " };
        let dense = if kind == "ssn" {
            "123-45-6789 ".repeat(1000)
        } else {
            "4000 0087 7000 0003 ".repeat(850)
        };
        vec![
            ("sparse ascii 74KB".into(), sparse(filler, 1100)),
            ("sparse unicode 78KB".into(), sparse(ufiller, 1500)),
            ("dense".into(), dense),
            ("digit noise 48KB".into(), noise),
            ("isolated digit /17".into(), isolated(17)),
            ("isolated digit /24".into(), isolated(24)),
            ("isolated digit /40".into(), isolated(40)),
            ("long digit run 60KB".into(), "9".repeat(60_000)),
            ("all non-digit 60KB".into(), "ab-cd ".repeat(10_000)),
            (
                "sparse head, dense tail".into(),
                format!("{}{}", "filler text ".repeat(40), dense_unit.repeat(1500)),
            ),
            (
                "dense head, sparse tail".into(),
                format!("{}{}", dense_unit.repeat(40), "filler text ".repeat(5000)),
            ),
        ]
    }

    fn time(f: &dyn Fn() -> usize) -> f64 {
        let mut reps = 1u32;
        loop {
            let t = Instant::now();
            let mut sink = 0;
            for _ in 0..reps {
                sink += std::hint::black_box(f());
            }
            let e = t.elapsed();
            std::hint::black_box(sink);
            if e.as_millis() >= 20 || reps >= 1_000_000 {
                return e.as_secs_f64() * 1e6 / f64::from(reps);
            }
            reps *= 2;
        }
    }
    fn median(mut v: Vec<f64>) -> f64 {
        v.sort_by(|a, b| a.partial_cmp(b).unwrap());
        v[v.len() / 2]
    }

    #[test]
    #[ignore]
    fn measure_1169_production() {
        for kind in ["ssn", "pan"] {
            for (name, input) in workloads(kind) {
                let i = input.as_str();
                let mut samples = Vec::new();
                for _ in 0..41 {
                    samples.push(time(&move || {
                        if kind == "ssn" {
                            pii_us_ssn::detect_us_ssns(i).len()
                        } else {
                            pii_payment_card::detect_payment_cards(i).len()
                        }
                    }));
                }
                println!(
                    "{kind} {name:22} production median {:9.2} us   min {:9.2} us",
                    median(samples.clone()),
                    samples.iter().copied().fold(f64::INFINITY, f64::min)
                );
            }
        }
    }
}
