+/// Inputs shorter than this are never classified as sparse: the dispatch
+/// itself would cost more than a short scan (issue #1169).
+const DIGIT_SPARSE_MIN_BYTES: usize = 256;
+/// Leading bytes counted by [`sparse_digit_head`].
+const DIGIT_HEAD_BYTES: usize = 128;
+/// A head with fewer ASCII digits than this is sparse.
+const DIGIT_HEAD_SPARSE_BELOW: usize = 8;
+/// Bytes [`next_ascii_digit`] tests at once.
+const DIGIT_BLOCK_BYTES: usize = 16;
+/// Bytes the original loop walks after each jump in a sparse input.
+const DIGIT_WINDOW_BYTES: usize = 32;
+
+/// `true` when `bytes` is long enough and its first [`DIGIT_HEAD_BYTES`] hold
+/// fewer than [`DIGIT_HEAD_SPARSE_BELOW`] ASCII digits.
+///
+/// This is the adaptive part of issue #1169. The digit-family detectors scan
+/// a digit-dense or digit-noisy input with the original byte loop, which
+/// every skipping variant measured slower there; only an input whose head
+/// looks sparse jumps with [`next_ascii_digit`] and runs the same loop over a
+/// [`DIGIT_WINDOW_BYTES`] window after each jump. The skipped bytes are
+/// non-digits, which the loop steps over anyway, so the choice affects time
+/// only, never the findings.
+#[inline]
+fn sparse_digit_head(bytes: &[u8]) -> bool {
+    bytes.len() >= DIGIT_SPARSE_MIN_BYTES
+        && bytes[..DIGIT_HEAD_BYTES]
+            .iter()
+            .filter(|byte| byte.is_ascii_digit())
+            .count()
+            < DIGIT_HEAD_SPARSE_BELOW
+}
+
+/// The offset of the first ASCII digit at or after `from`, or `bytes.len()`
+/// when there is none. Whole blocks without a digit are skipped, then the
+/// block holding the first digit is walked byte by byte, so the result is
+/// always the exact first digit.
+fn next_ascii_digit(bytes: &[u8], from: usize) -> usize {
+    let mut at = from.min(bytes.len());
+    while let Some(block) = bytes.get(at..at + DIGIT_BLOCK_BYTES) {
+        if block
+            .iter()
+            .fold(false, |found, byte| found | byte.is_ascii_digit())
+        {
+            break;
+        }
+        at += DIGIT_BLOCK_BYTES;
+    }
+    while at < bytes.len() && !bytes[at].is_ascii_digit() {
+        at += 1;
+    }
+    at
+}

+            "\u{e9}",

+            "\u{200b}",
+            "\u{301}",
+            "card: ",
+            "\u{ce74}\u{b4dc}\u{bc88}\u{d638}: ",
+        ];
+        let mut rng = XorShift32::new(0x1169_0004);
+        let mut inputs = Vec::new();
+        for _ in 0..3_000 {
+            let mut text = String::new();
+            for _ in 0..rng.below(10) {
+                // Gaps straddle the 16-byte probe and block sizes.
+                text.push_str(&" ".repeat([0, 1, 7, 15, 16, 17, 31, 33][rng.below(8)]));
+                text.push_str(PIECES[rng.below(PIECES.len())]);
+            }
+            inputs.push(text);
+        }
+        for _ in 0..1_500 {
+            inputs.push(rng.text(BOUNDARY_PIECES, 24));
+        }
+        inputs.push("9".repeat(1_000));
+        // Long enough to be classified by head density: sparse head with a
+        // dense tail, dense head with a sparse tail, and a sparse head only.
+        inputs.push("filler text ".repeat(40) + &"4000 0087 7000 0003 ".repeat(30));
+        inputs.push("4000 0087 7000 0003 ".repeat(30) + &"filler text ".repeat(40));
+        inputs.push("filler text ".repeat(60) + "4000 0087 7000 0003");
+        inputs.push("a".repeat(500) + "4000008770000003" + &"b".repeat(500));
+        inputs
     }

+    #[test]
+    fn adaptive_discovery_equals_the_byte_walk() {
+        let mut found = 0;
+        for input in corpus() {
+            let old = oracle_detect_payment_cards(&input);
+            for new in [
+                detect_payment_cards(&input),
+                collect_payment_cards(&input, true),
+                collect_payment_cards(&input, false),
+            ] {
+                assert_eq!(format!("{new:?}"), format!("{old:?}"), "{input:?}");
+            }
+            found += old.len();
+        }
+        assert!(found > 500, "only {found} candidates");
+    }

+    #[test]
+    fn every_window_of_a_sparse_input_matches_the_byte_walk() {
+        let input = format!(
+            "{}4000 0087 7000 0003{}\u{e9}4111111111111111 {}",
+            "a".repeat(40),
+            "b".repeat(33),
+            "c".repeat(20)
+        );
+        for start in 0..=input.len() {
+            for end in start..=input.len() {
+                if input.is_char_boundary(start) && input.is_char_boundary(end) {
+                    let window = &input[start..end];
+                    let expected = format!("{:?}", oracle_detect_payment_cards(window));
+                    assert_eq!(
+                        format!("{:?}", detect_payment_cards(window)),
+                        expected,
+                        "{window:?}"
+                    );
+                    assert_eq!(
+                        format!("{:?}", collect_payment_cards(window, true)),
+                        expected,
+                        "{window:?}"
+                    );
+                }
+            }
+        }
+    }
